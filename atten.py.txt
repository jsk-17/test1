"""스터디 출석부 프로그램
- 날짜와 이름을 입력하면 출석이 기록됩니다.
- 출석 목록을 볼 수 있습니다.
- 외부 라이브러리 없이 파이썬 기본 기능만 사용합니다.
- 출석 기록은 attendance.csv 파일에 저장되어 프로그램을 꺼도 남습니다.
"""

import csv
import os
from datetime import date

# 출석 기록을 저장할 파일 이름
FILE_NAME = "attendance.csv"


def add_attendance():
    """출석 추가: 날짜와 이름을 입력받아 파일에 저장"""
    # 날짜를 비워 두면 오늘 날짜를 자동으로 사용
    day = input("날짜를 입력하세요 (예: 2024-05-20, 비우면 오늘): ").strip()
    if day == "":
        day = str(date.today())

    name = input("이름을 입력하세요: ").strip()
    if name == "":
        print("이름이 비어 있어요. 다시 시도해 주세요.\n")
        return

    # 파일을 '추가 모드(a)'로 열어서 한 줄씩 쌓음
    with open(FILE_NAME, "a", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow([day, name])

    print(f"기록 완료! [{day}] {name}\n")


def show_attendance():
    """출석 목록 보기: 파일을 읽어서 화면에 출력"""
    # 파일이 아직 없으면 기록이 없는 것
    if not os.path.exists(FILE_NAME):
        print("아직 출석 기록이 없어요.\n")
        return

    with open(FILE_NAME, "r", newline="", encoding="utf-8") as f:
        rows = list(csv.reader(f))

    if not rows:
        print("아직 출석 기록이 없어요.\n")
        return

    print("\n===== 출석 목록 =====")
    for i, row in enumerate(rows, start=1):
        # row[0]=날짜, row[1]=이름
        print(f"{i}. {row[0]}  {row[1]}")
    print(f"총 {len(rows)}건\n")


def main():
    """메뉴를 보여 주고 사용자의 선택을 받는 중심 함수"""
    while True:
        print("===== 스터디 출석부 =====")
        print("1. 출석 기록하기")
        print("2. 출석 목록 보기")
        print("3. 종료")
        choice = input("번호를 선택하세요: ").strip()

        if choice == "1":
            add_attendance()
        elif choice == "2":
            show_attendance()
        elif choice == "3":
            print("출석부를 종료합니다. 수고하셨어요!")
            break
        else:
            print("1, 2, 3 중에서 골라 주세요.\n")


if __name__ == "__main__":
    main()
